这节学习正常内存块归还deallocate，这次来学习它的源码

```c++
  static void deallocate(void* __p, size_t __n)
  {                 //要归还的内存块的起始地址，字节数
    if (__n > (size_t) _MAX_BYTES)//如果归还的字节>128，大块内存直接free 
      malloc_alloc::deallocate(__p, __n);
    else {//从内存池分配的
      _Obj* __STL_VOLATILE*  __my_free_list
          = _S_free_list + _S_freelist_index(__n);
        //找到在自由链表中的位置
      _Obj* __q = (_Obj*)__p;
     //将归还内存块的地址(万能指针）赋给 __q（Obj*)
      // acquire lock
#       ifndef _NOTHREADS//线程安全操作
      /*REFERENCED*/
      _Lock __lock_instance;
#       endif /* _NOTHREADS */
      __q -> _M_free_list_link = *__my_free_list;
      *__my_free_list = __q;//进行链表头插操作
      // lock is released here
    }
  }
```

```c++
  static void deallocate(void* __p, size_t __n)
  {                 //要归还的内存块的起始地址，字节数
    if (__n > (size_t) _MAX_BYTES)//如果归还的字节>128，大块内存直接free 
      malloc_alloc::deallocate(__p, __n);
    else {//从内存池分配的
      _Obj* __STL_VOLATILE*  __my_free_list
          = _S_free_list + _S_freelist_index(__n);
        //找到在自由链表中的位置
      _Obj* __q = (_Obj*)__p;
     //将归还内存块的地址(万能指针）赋给 __q（Obj*)
      // acquire lock
#       ifndef _NOTHREADS//线程安全操作
      /*REFERENCED*/
      _Lock __lock_instance;
#       endif /* _NOTHREADS */
      __q -> _M_free_list_link = *__my_free_list;
      *__my_free_list = __q;//进行链表头插操作
      // lock is released here
    }
  }
```

reallocate扩容/缩容函数

```c++
template <bool threads, int inst>
void*                       //chunk块起始地址，原来chunk大小，新的chunk大小
__default_alloc_template<threads, inst>::reallocate(void* __p,
                                                    size_t __old_sz,
                                                    size_t __new_sz)
{
    void* __result;
    size_t __copy_sz;
    //如果新值旧值都大于128，说明不是从内存池中分配的，使用realloc调整大小
    if (__old_sz > (size_t) _MAX_BYTES && __new_sz > (size_t) _MAX_BYTES) {
        return(realloc(__p, __new_sz));
    }
    //如果是小块内存，如果新值旧值向上取值结果一样，没有必要调整，直接返回
    if (_S_round_up(__old_sz) == _S_round_up(__new_sz)) return(__p);
    //如果新值旧值不同，通过allocate分配对应new值的大小
    __result = allocate(__new_sz);
    //如果new>old，就拷贝old大小的数据，否则拷贝new大小的数据
    __copy_sz = __new_sz > __old_sz? __old_sz : __new_sz;
    memcpy(__result, __p, __copy_sz);//使用memcpy把数据从旧的chunk块copy到新的chunk块
    deallocate(__p, __old_sz);
    //释放掉旧的chunk块，返回新chunk块的地址
    return(__result);
}
```
